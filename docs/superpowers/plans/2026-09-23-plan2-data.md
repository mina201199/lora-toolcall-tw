`scripts/check_template.py`：

```python
"""Verify that our hand-rendered prompt equals the model's own chat template.

Needs a running llama-server (started with --jinja). For each scenario it asks
the server to render two requests via POST /apply-template:

  A. the "official" way: `tools` field + structured assistant `tool_calls`
  B. our way: tools rendered into the system text, tool calls as plain text

and diffs the two prompts. Any difference means training and serving would
see different text, so fix `prompting.py` / `parsing.render_tool_calls`
before going further.

    python scripts/check_template.py --server http://localhost:8080
    python scripts/check_template.py --save tests/fixtures   # record for CI
"""

from __future__ import annotations

import argparse
import difflib
import json
import sys
from datetime import date
from pathlib import Path

import httpx

from twagent.parsing import ToolCall, render_tool_calls
from twagent.prompting import build_persona, build_system_prompt
from twagent.tools.schemas import TOOLS

TODAY = date(2026, 9, 23)
CALL = ToolCall("get_weather", {"city": "臺中市", "date": "2026-09-24"})
CALL_2 = ToolCall("get_weather", {"city": "高雄市", "date": "2026-09-24"})
RESULT = json.dumps({"city": "臺中市", "condition": "多雲"}, ensure_ascii=False)
RESULT_2 = json.dumps({"city": "高雄市", "condition": "晴"}, ensure_ascii=False)


def _official_call(calls: list[ToolCall], content: str = "") -> dict:
    return {
        "role": "assistant",
        "content": content,
        "tool_calls": [
            {
                "id": f"call_{i}",
                "type": "function",
                "function": {
                    "name": c.name,
                    "arguments": json.dumps(c.arguments, ensure_ascii=False),
                },
            }
            for i, c in enumerate(calls)
        ],
    }


def _ours_call(calls: list[ToolCall], content: str = "") -> dict:
    rendered = render_tool_calls(calls)
    return {"role": "assistant", "content": f"{content}\n{rendered}" if content else rendered}


def scenarios() -> dict[str, tuple[dict, dict]]:
    user = {"role": "user", "content": "明天台中會下雨嗎"}
    official_system = {"role": "system", "content": build_persona(TODAY)}
    ours_system = {"role": "system", "content": build_system_prompt(TODAY)}

    def tool(content: str, i: int, official: bool) -> dict:
        msg = {"role": "tool", "content": content}
        if official:
            msg["tool_call_id"] = f"call_{i}"
        return msg

    both = [CALL, CALL_2]
    answer = {"role": "assistant", "content": "明天臺中多雲。"}
    follow_up = {"role": "user", "content": "那後天呢？"}
    return {
        "first_turn": (
            {"messages": [official_system, user], "tools": TOOLS},
            {"messages": [ours_system, user]},
        ),
        "after_tool_call": (
            {
                "messages": [official_system, user, _official_call([CALL]), tool(RESULT, 0, True)],
                "tools": TOOLS,
            },
            {"messages": [ours_system, user, _ours_call([CALL]), tool(RESULT, 0, False)]},
        ),
        "parallel_calls": (
            {
                "messages": [
                    official_system,
                    user,
                    _official_call(both),
                    tool(RESULT, 0, True),
                    tool(RESULT_2, 1, True),
                ],
                "tools": TOOLS,
            },
            {
                "messages": [
                    ours_system,
                    user,
                    _ours_call(both),
                    tool(RESULT, 0, False),
                    tool(RESULT_2, 1, False),
                ]
            },
        ),
        "text_call_answer_followup": (
            {
                "messages": [
                    official_system,
                    user,
                    _official_call([CALL], "我幫你查一下。"),
                    tool(RESULT, 0, True),
                    answer,
                    follow_up,
                ],
                "tools": TOOLS,
            },
            {
                "messages": [
                    ours_system,
                    user,
                    _ours_call([CALL], "我幫你查一下。"),
                    tool(RESULT, 0, False),
                    answer,
                    follow_up,
                ]
            },
        ),
    }


def render(http: httpx.Client, server: str, body: dict) -> str:
    resp = http.post(f"{server.rstrip('/')}/apply-template", json=body)
    resp.raise_for_status()
    return resp.json()["prompt"]


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--server", default="http://localhost:8080")
    ap.add_argument(
        "--save", type=Path, help="write each official rendering to DIR/qwen_<name>.txt"
    )
    args = ap.parse_args()
    failed = False
    with httpx.Client(timeout=30) as http:
        for name, (official, ours) in scenarios().items():
            a, b = render(http, args.server, official), render(http, args.server, ours)
            if args.save:
                args.save.mkdir(parents=True, exist_ok=True)
                (args.save / f"qwen_{name}.txt").write_text(a, encoding="utf-8", newline="")
            if a == b:
                print(f"[same] {name}")
                continue
            failed = True
            print(f"[DIFF] {name}")
            sys.stdout.writelines(
                difflib.unified_diff(
                    a.splitlines(keepends=True), b.splitlines(keepends=True), "official", "ours"
                )
            )
    return 1 if failed else 0


if __name__ == "__main__":
    sys.exit(main())
```

`src/twagent/render.py`：

```python
"""Qwen chat-template rendering, done by us, with loss-mask boundaries.

Training (plan 3) tokenises the segments this module returns instead of calling
`tokenizer.apply_chat_template`, so the text the model is trained on is exactly
the text `tests/fixtures/qwen_*.txt` recorded from the model's own template --
the tests below the module prove it byte for byte.

Only assistant content (plus its closing `<|im_end|>`) is marked `train=True`.
System, user and tool-result text is context: the model should read it, never
learn to produce it.

Consecutive `tool` messages are merged into one user turn, each wrapped in
`<tool_response>` -- that is how the Qwen template renders them.
"""

from __future__ import annotations

from dataclasses import dataclass

from twagent.llm import Message

IM_START = "<|im_start|>"
IM_END = "<|im_end|>"


@dataclass(frozen=True)
class Segment:
    text: str
    train: bool


def render_chat(messages: list[Message], *, add_generation_prompt: bool = False) -> list[Segment]:
    segments: list[Segment] = []
    i = 0
    while i < len(messages):
        role, content = messages[i]["role"], messages[i]["content"]
        if role in ("system", "user"):
            segments.append(Segment(f"{IM_START}{role}\n{content}{IM_END}\n", train=False))
        elif role == "assistant":
            segments.append(Segment(f"{IM_START}assistant\n", train=False))
            segments.append(Segment(f"{content}{IM_END}", train=True))
            segments.append(Segment("\n", train=False))
        elif role == "tool":
            responses = []
            while i < len(messages) and messages[i]["role"] == "tool":
                responses.append(f"\n<tool_response>\n{messages[i]['content']}\n</tool_response>")
                i += 1
            segments.append(Segment(f"{IM_START}user{''.join(responses)}{IM_END}\n", train=False))
            continue
        else:
            raise ValueError(f"unknown role {role!r}")
        i += 1
    if add_generation_prompt:
        segments.append(Segment(f"{IM_START}assistant\n", train=False))
    return segments


def to_text(segments: list[Segment]) -> str:
    return "".join(s.text for s in segments)
```

```python
"""Typed view of configs/*.yaml.

Unknown keys are an error rather than silently ignored: a typo in a config
file must not produce a run that looks configured but is not.
"""

from __future__ import annotations

from dataclasses import asdict, dataclass, field, fields
from pathlib import Path
from typing import Any

import yaml


@dataclass(frozen=True)
class AgentConfig:
    max_steps: int = 5
    parse_retries: int = 1
    schema_retries: int = 2


@dataclass(frozen=True)
class LLMConfig:
    base_url: str = "http://localhost:8080/v1"
    model: str = "qwen"
    temperature: float = 0.0
    max_tokens: int = 512
    timeout_s: float = 60.0


@dataclass(frozen=True)
class ToolsConfig:
    mode: str = "mock"


@dataclass(frozen=True)
class ModelPrice:
    """USD per 1M tokens."""

    input: float
    output: float
    cache_read: float
    cache_write: float


@dataclass(frozen=True)
class SynthConfig:
    model: str = "claude-sonnet-5"
    fallback_model: str = "claude-haiku-4-5"
    budget_usd: float = 15.0
    train_target: int = 3000
    query_batch_size: int = 10
    max_tokens: int = 1024
    disable_thinking: bool = True
    seed: int = 3407
    prices: dict[str, ModelPrice] = field(default_factory=dict)


@dataclass(frozen=True)
class Config:
    agent: AgentConfig
    llm: LLMConfig
    tools: ToolsConfig
    synth: SynthConfig = field(default_factory=SynthConfig)

    def to_dict(self) -> dict[str, Any]:
        return asdict(self)


def _build(cls: type, section: str, raw: dict[str, Any] | None) -> Any:
    raw = raw or {}
    allowed = {f.name for f in fields(cls)}
    unknown = set(raw) - allowed
    if unknown:
        raise ValueError(f"unknown key(s) in [{section}]: {sorted(unknown)}")
    return cls(**raw)


def _build_synth(raw: dict[str, Any] | None) -> SynthConfig:
    raw = dict(raw or {})
    prices = {
        model: _build(ModelPrice, f"synth.prices.{model}", price)
        for model, price in (raw.pop("prices", None) or {}).items()
    }
    return _build(SynthConfig, "synth", {**raw, "prices": prices})


def load_config(path: str | Path) -> Config:
    with open(path, encoding="utf-8") as f:
        raw = yaml.safe_load(f) or {}
    unknown = set(raw) - {"agent", "llm", "tools", "synth"}
    if unknown:
        raise ValueError(f"unknown section(s): {sorted(unknown)}")
    return Config(
        agent=_build(AgentConfig, "agent", raw.get("agent")),
        llm=_build(LLMConfig, "llm", raw.get("llm")),
        tools=_build(ToolsConfig, "tools", raw.get("tools")),
        synth=_build_synth(raw.get("synth")),
    )
```

`src/twagent/claude.py`：

```python
"""Claude as an `LLMClient`: the data-synthesis teacher, and later the B2
baseline and the serving fallback.

Claude does not get Anthropic `tools`. It gets the same system prompt Qwen
gets -- which already contains the `<tool_call>` instructions -- and answers
in plain text, so its output goes through the same `parse_output` as every
other model. Tool results are sent back the way the Qwen template renders
them: `<tool_response>` blocks inside a user turn.

For prompt caching the system prompt is split in two: the tool block (large,
identical for every request) goes first and carries `cache_control`; the
persona (small, contains today's date) follows it.
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any

import anthropic

from twagent.config import ModelPrice
from twagent.llm import LLMError, Message

TOOLS_MARKER = "\n\n# Tools\n"


@dataclass
class Usage:
    requests: int = 0
    input_tokens: int = 0
    output_tokens: int = 0
    cache_read_input_tokens: int = 0
    cache_creation_input_tokens: int = 0

    def add(self, usage: Any) -> None:
        self.requests += 1
        self.input_tokens += usage.input_tokens or 0
        self.output_tokens += usage.output_tokens or 0
        self.cache_read_input_tokens += getattr(usage, "cache_read_input_tokens", 0) or 0
        self.cache_creation_input_tokens += getattr(usage, "cache_creation_input_tokens", 0) or 0

    def cost_usd(self, price: ModelPrice) -> float:
        return (
            self.input_tokens * price.input
            + self.output_tokens * price.output
            + self.cache_read_input_tokens * price.cache_read
            + self.cache_creation_input_tokens * price.cache_write
        ) / 1_000_000


def _system_blocks(system: str) -> list[dict[str, Any]]:
    if TOOLS_MARKER not in system:
        return [{"type": "text", "text": system}]
    persona, tools = system.split(TOOLS_MARKER, 1)
    return [
        {"type": "text", "text": "# Tools\n" + tools, "cache_control": {"type": "ephemeral"}},
        {"type": "text", "text": persona},
    ]


def to_anthropic_request(messages: list[Message]) -> tuple[list[dict[str, Any]], list[dict]]:
    """Our messages -> (system blocks, Anthropic messages).

    Tool results become `<tool_response>` text in a user turn, and consecutive
    same-role turns are merged so roles strictly alternate.
    """
    system: list[dict[str, Any]] = []
    out: list[dict[str, str]] = []
    for m in messages:
        role, content = m["role"], m["content"]
        if role == "system":
            system = _system_blocks(content)
            continue
        if role == "tool":
            role, content = "user", f"<tool_response>\n{content}\n</tool_response>"
        elif role not in ("user", "assistant"):
            raise ValueError(f"unknown role {role!r}")
        if out and out[-1]["role"] == role:
            out[-1]["content"] += "\n" + content
        else:
            out.append({"role": role, "content": content})
    return system, out


class AnthropicClient:
    """`LLMClient` backed by the Messages API. `usage` accumulates every call."""

    def __init__(
        self,
        model: str,
        *,
        max_tokens: int = 1024,
        disable_thinking: bool = True,
        client: Any = None,
    ) -> None:
        self.model = model
        self.max_tokens = max_tokens
        self.disable_thinking = disable_thinking
        self.usage = Usage()
        self._client = client if client is not None else anthropic.Anthropic()

    def complete(self, messages: list[Message]) -> str:
        system, msgs = to_anthropic_request(messages)
        kwargs: dict[str, Any] = {
            "model": self.model,
            "max_tokens": self.max_tokens,
            "messages": msgs,
        }
        if system:
            kwargs["system"] = system
        if self.disable_thinking:
            kwargs["thinking"] = {"type": "disabled"}
        try:
            response = self._client.messages.create(**kwargs)
        except anthropic.APIError as e:
            raise LLMError(f"{self.model}: {e}") from e
        self.usage.add(response.usage)
        if response.stop_reason == "refusal":
            raise LLMError(f"{self.model} refused the request")
        if response.stop_reason == "max_tokens":
            raise LLMError(f"{self.model} hit max_tokens={self.max_tokens}")
        return "".join(b.text for b in response.content if b.type == "text")
```

```python
"""The scenario matrix every synthetic sample is drawn from (spec §5.1-5.2).

A scenario fixes what a sample must be -- category, tools, speaking style,
the relative-date phrase to use and the "today" it is relative to -- before
any model writes a word. Validation later checks the teacher's trajectory
against the scenario, so a sample that drifted from its brief is dropped.

`group` is the unit of train/valid/test splitting: every sample of one
(category, tools, style) cell lands in the same split, so the test set only
contains combinations the model never trained on.
"""

from __future__ import annotations

import random
from dataclasses import asdict, dataclass
from datetime import date, timedelta
from typing import Any

CATEGORY_SHARE = {
    "single": 0.35,
    "parallel": 0.15,
    "sequential": 0.15,
    "clarify": 0.15,
    "no_tool": 0.15,
    "tool_error": 0.05,
}

STYLES = (
    "口語隨興",
    "有錯字或注音文",
    "中英夾雜",
    "很長而且繞圈子",
    "非常簡短",
    "客氣正式",
)

_ALL_TOOLS = (
    "get_weather",
    "search_thsr",
    "search_tra",
    "find_youbike",
    "check_invoice",
    "calendar_add",
    "calendar_list",
)

COMBOS: dict[str, list[tuple[str, ...]]] = {
    "single": [(t,) for t in _ALL_TOOLS],
    "parallel": [
        ("get_weather", "get_weather"),
        ("search_thsr", "get_weather"),
        ("find_youbike", "get_weather"),
        ("check_invoice", "check_invoice"),
        ("calendar_list", "get_weather"),
        ("search_tra", "search_thsr"),
    ],
    # Later steps need the earlier step's result (spec §5.1).
    "sequential": [
        ("search_thsr", "calendar_add"),
        ("search_tra", "calendar_add"),
        ("calendar_list", "calendar_add"),
        ("get_weather", "calendar_add"),
    ],
    "clarify": [(t,) for t in _ALL_TOOLS],
    "no_tool": [()],
    "tool_error": [(t,) for t in _ALL_TOOLS if t != "calendar_list"],
}

NO_TOOL_KINDS = ("閒聊", "一般常識問題", "不支援的請求（訂票、付款、叫車等）")

DATED_TOOLS = {"get_weather", "search_thsr", "search_tra", "calendar_add", "calendar_list"}
DATE_EXPRS = ("今天", "明天", "後天", "大後天", "這週五", "下週一", "下禮拜三", "週六", "3天後")
FAILURES = ("連線逾時", "服務暫時無法使用", "上游資料格式錯誤")


@dataclass(frozen=True)
class Scenario:
    id: str
    category: str
    tools: tuple[str, ...]
    style: str
    today: date
    date_expr: str | None = None
    kind: str = ""  # no_tool only
    failure: str = ""  # tool_error only: message the first tool fails with

    @property
    def group(self) -> str:
        what = "+".join(self.tools) if self.tools else self.kind
        return f"{self.category}|{what}|{self.style}"

    def to_dict(self) -> dict[str, Any]:
        d = asdict(self)
        d["tools"] = list(self.tools)
        d["today"] = self.today.isoformat()
        return d

    @classmethod
    def from_dict(cls, d: dict[str, Any]) -> Scenario:
        return cls(**{**d, "tools": tuple(d["tools"]), "today": date.fromisoformat(d["today"])})


def _counts(total: int) -> dict[str, int]:
    counts = {c: int(total * share) for c, share in CATEGORY_SHARE.items()}
    counts["single"] += total - sum(counts.values())  # rounding remainder
    return counts


def build_scenarios(total: int, seed: int) -> list[Scenario]:
    """`total` scenarios in CATEGORY_SHARE proportions, deterministic in `seed`.

    Within a category, cells (combo x style, or kind x style) are visited
    round-robin in a shuffled order, so every cell gets used before any cell
    is used twice.
    """
    rng = random.Random(seed)
    year_start = date(2026, 1, 1)
    out: list[Scenario] = []
    for category, n in _counts(total).items():
        if category == "no_tool":
            cells = [((), kind, style) for kind in NO_TOOL_KINDS for style in STYLES]
        else:
            cells = [(combo, "", style) for combo in COMBOS[category] for style in STYLES]
        rng.shuffle(cells)
        for i in range(n):
            tools, kind, style = cells[i % len(cells)]
            dated = any(t in DATED_TOOLS for t in tools)
            out.append(
                Scenario(
                    id=f"s{len(out):05d}",
                    category=category,
                    tools=tools,
                    style=style,
                    today=year_start + timedelta(days=rng.randrange(365)),
                    date_expr=rng.choice(DATE_EXPRS) if dated else None,
                    kind=kind,
                    failure=rng.choice(FAILURES) if category == "tool_error" else "",
                )
            )
    return out
```

`src/twagent/data/synth.py`：

```python
"""Two-stage synthesis (spec §5.2, as revised for plan 2):

1. A generator model writes one user message per scenario (batched).
2. A teacher model runs through the real agent loop -- `run_agent` with
   `MockTools` and the exact Qwen system prompt -- and its trajectory becomes
   the training sample. Tool results always come from the mocks, never from
   the model, and a sequential step naturally sees the previous result.

The teacher gets no second chances (`parse_retries=0`, `schema_retries=0`):
a trajectory with any error is recorded and later dropped by validation, so
the training data never shows a model recovering from its own bad output.

Output is append-only JSONL, one record per scenario, so an interrupted run
resumes where it stopped.
"""

from __future__ import annotations

import json
from collections.abc import Callable, Iterable
from dataclasses import dataclass
from pathlib import Path
from typing import Any

from twagent.agent import run_agent
from twagent.config import AgentConfig
from twagent.dates import format_today
from twagent.llm import LLMClient, LLMError
from twagent.prompting import build_system_prompt
from twagent.tools.mock import MockTools
from twagent.tools.schemas import get_tool

from .scenarios import Scenario

TEACHER_CONFIG = AgentConfig(max_steps=5, parse_retries=0, schema_retries=0)

_CATEGORY_BRIEF = {
    "single": "只需要呼叫一次下列工具就能完成，而且該工具所需的資訊都已經說清楚。",
    "parallel": "一句話裡同時要兩件互不相依的事，分別對應下列工具（同一個工具出現兩次代表兩組不同參數，例如兩個城市）。資訊都要齊全。",
    "sequential": "必須先用第一個工具查到結果，再根據查到的內容使用第二個工具（例如查到班次後，把搭乘時間排進行事曆；看了天氣後決定要不要排行程）。資訊都要齊全。",
    "clarify": "想用下列工具，但刻意漏掉至少一個必要資訊（例如沒說起訖站、沒說日期、沒給發票號碼、沒說行程時間），讓助理必須先追問。",
    "no_tool": "完全不需要任何工具。類型：{kind}。",
    "tool_error": "一般的單一工具需求，資訊齊全（使用者不知道工具之後會故障）。",
}

_RULES = """規則：
- 使用繁體中文與臺灣用語，像真人在手機上隨手打字；依指定的風格寫。
- 不可以出現工具的英文名稱，也不要提到「工具」「API」。
- 地名、車站可以用口語說法（台北、高雄、台大附近）。
- 有指定「日期說法」時，必須原封不動使用那個說法，不要自己寫出具體日期。
- 發票對獎要附上 8 碼號碼，並用「這一期」「上一期」或月份描述期別。
- 新增行程要講清楚事由與時間（開始與結束，或開始與持續多久）。
- 每則 5 到 80 個字；風格是「很長而且繞圈子」時可以到 150 字。
- 只輸出一個 JSON 陣列，不要任何其他文字。每個元素為 {"id": "...", "message": "..."}，順序與清單相同。"""


def _describe(s: Scenario) -> str:
    brief = _CATEGORY_BRIEF[s.category].format(kind=s.kind)
    parts = [f"id：{s.id}", f"需求：{brief}", f"風格：{s.style}", f"今天：{format_today(s.today)}"]
    if s.tools:
        tools = "、".join(f"{t}（{get_tool(t)['function']['description']}）" for t in s.tools)
        parts.append(f"工具：{tools}")
    if s.date_expr:
        parts.append(f"日期說法：必須使用「{s.date_expr}」")
    return "- " + "｜".join(parts)


def build_query_prompt(batch: list[Scenario]) -> str:
    lines = "\n".join(_describe(s) for s in batch)
    return (
        "請替一個臺灣生活助理的訓練資料，寫出使用者傳給助理的第一則訊息。"
        "每個情境寫一則。\n\n"
        f"{_RULES}\n\n情境清單：\n{lines}"
    )


def parse_query_response(raw: str, batch: list[Scenario]) -> dict[str, str]:
    """Scenario id -> message. Anything malformed or unknown is left out."""
    start, end = raw.find("["), raw.rfind("]")
    if start == -1 or end <= start:
        return {}
    try:
        items = json.loads(raw[start : end + 1])
    except json.JSONDecodeError:
        return {}
    wanted = {s.id for s in batch}
    out: dict[str, str] = {}
    for item in items if isinstance(items, list) else []:
        if not isinstance(item, dict):
            continue
        sid, msg = item.get("id"), item.get("message")
        if sid in wanted and isinstance(msg, str) and msg.strip():
            out[sid] = msg.strip()
    return out


def run_trajectory(scenario: Scenario, message: str, teacher: LLMClient) -> dict[str, Any]:
    failures = {scenario.tools[0]: scenario.failure} if scenario.failure else None
    tools = MockTools(failures=failures)
    result = run_agent(
        [{"role": "user", "content": message}],
        llm=teacher,
        tools=tools,
        system_prompt=build_system_prompt(scenario.today),
        config=TEACHER_CONFIG,
    )
    return {
        "scenario": scenario.to_dict(),
        "message": message,
        "result": result.to_dict(),
        "messages": result.messages,
        "calendar": tools.events(),
    }


@dataclass
class SynthReport:
    written: int = 0
    already_done: int = 0
    missing_queries: int = 0
    stopped: str = "complete"  # complete | budget


def load_records(path: Path) -> list[dict[str, Any]]:
    if not path.exists():
        return []
    with open(path, encoding="utf-8") as f:
        return [json.loads(line) for line in f if line.strip()]


def _chunks(items: list[Scenario], size: int) -> Iterable[list[Scenario]]:
    for i in range(0, len(items), size):
        yield items[i : i + size]


def synthesize(
    scenarios: list[Scenario],
    *,
    query_llm: LLMClient,
    teacher: LLMClient,
    out_path: Path,
    batch_size: int,
    spent_usd: Callable[[], float],
    budget_usd: float,
) -> SynthReport:
    """Append one record per scenario to `out_path`; stop before exceeding budget."""
    done = {r["scenario"]["id"] for r in load_records(out_path)}
    report = SynthReport(already_done=len(done))
    todo = [s for s in scenarios if s.id not in done]
    out_path.parent.mkdir(parents=True, exist_ok=True)
    with open(out_path, "a", encoding="utf-8") as f:
        for batch in _chunks(todo, batch_size):
            if spent_usd() >= budget_usd:
                report.stopped = "budget"
                break
            try:
                raw = query_llm.complete([{"role": "user", "content": build_query_prompt(batch)}])
            except LLMError:
                report.missing_queries += len(batch)
                continue
            messages = parse_query_response(raw, batch)
            for s in batch:
                if s.id not in messages:
                    report.missing_queries += 1
                    continue
                record = run_trajectory(s, messages[s.id], teacher)
                f.write(json.dumps(record, ensure_ascii=False) + "\n")
                f.flush()
                report.written += 1
    return report
```

`src/twagent/data/validate.py`：

```python
"""Decide whether one teacher trajectory is good enough to train on.

`validate_record` returns the reasons a record is rejected; an empty list
keeps it. Every reason is short and stable so `data_report.json` can count
them. Checks, in the spirit of astro-vlm-lora's "one number that does not fit":

- the run ended normally and the teacher never erred (no parse/schema/llm
  error, no fallback) -- we do not train on recoveries;
- the calls match the scenario's category and tools;
- a relative date resolves to exactly what `dates.py` computes;
- no simplified characters or mainland wording (OpenCC s2twp round trip);
- the generated user message does not leak tool names.
"""

from __future__ import annotations

from collections import Counter
from typing import Any

from opencc import OpenCC

from twagent.dates import resolve_date_expr
from twagent.parsing import parse_output
from twagent.tools.schemas import tool_names

from .scenarios import Scenario

_BAD_EVENTS = {"parse_error", "schema_error", "llm_error", "fallback"}
_S2TWP = OpenCC("s2twp")


def wording_problem(text: str) -> str | None:
    """A snippet showing simplified/mainland wording, or None if clean.

    台 vs 臺 is not a problem (both are Taiwanese usage), so it is normalised
    before the comparison.
    """
    normalised = text.replace("台", "臺")
    converted = _S2TWP.convert(normalised)
    if converted == normalised:
        return None
    for i, (a, b) in enumerate(zip(normalised, converted, strict=False)):
        if a != b:
            return f"{normalised[max(0, i - 4) : i + 4]} -> {converted[max(0, i - 4) : i + 4]}"
    return f"{normalised[-8:]} -> {converted[-8:]}"


def _calls_per_turn(result: dict[str, Any]) -> list[list[str]]:
    turns = []
    for event in result["trace"]:
        if event["type"] == "llm_output":
            names = [c.name for c in parse_output(event["raw"]).tool_calls]
            if names:
                turns.append(names)
    return turns


def _category_problems(s: Scenario, result: dict[str, Any]) -> list[str]:
    names = [c["name"] for c in result["tool_calls"]]
    turns = _calls_per_turn(result)
    final = result["final_text"]
    if s.category == "single" and names != list(s.tools):
        return [f"single: expected {list(s.tools)}, got {names}"]
    if s.category == "parallel" and (len(turns) != 1 or Counter(turns[0]) != Counter(s.tools)):
        return [f"parallel: expected one turn with {list(s.tools)}, got {turns}"]
    if s.category == "sequential" and (names != list(s.tools) or len(turns) < 2):
        return [f"sequential: expected {list(s.tools)} over >=2 turns, got {turns}"]
    if s.category in ("clarify", "no_tool") and names:
        return [f"{s.category}: expected no calls, got {names}"]
    if s.category == "clarify" and not ("？" in final or "?" in final):
        return ["clarify: reply is not a question"]
    if s.category == "tool_error":
        if not names or names[0] != s.tools[0]:
            return [f"tool_error: expected {s.tools[0]} first, got {names}"]
        if not any(e["type"] == "tool_error" for e in result["trace"]):
            return ["tool_error: tool never failed"]
    return []


def _date_problem(s: Scenario, result: dict[str, Any]) -> str | None:
    if not s.date_expr:
        return None
    for call in result["tool_calls"]:
        args = call["arguments"]
        got = args.get("date") or (args.get("start") or "")[:10] or None
        if got is None:
            continue
        expected = resolve_date_expr(s.date_expr, s.today).isoformat()
        return None if got == expected else f"date: {s.date_expr} should be {expected}, got {got}"
    return None


def validate_record(record: dict[str, Any]) -> list[str]:
    s = Scenario.from_dict(record["scenario"])
    result = record["result"]
    if result["stopped_reason"] != "answer":
        return [f"stopped: {result['stopped_reason']}"]
    problems: list[str] = []
    bad = sorted({e["type"] for e in result["trace"]} & _BAD_EVENTS)
    if bad:
        problems.append(f"teacher error: {', '.join(bad)}")
    if not result["final_text"].strip():
        problems.append("empty final reply")
    problems += _category_problems(s, result)
    if (p := _date_problem(s, result)) is not None:
        problems.append(p)
    message = record["message"]
    if any(name in message for name in tool_names()):
        problems.append("tool name leaked into user message")
    texts = [message] + [m["content"] for m in record["messages"] if m["role"] == "assistant"]
    for text in texts:
        if (w := wording_problem(text)) is not None:
            problems.append(f"wording: {w}")
            break
    return problems


def reason_key(problem: str) -> str:
    """Collapse a problem string to its counting bucket (text before ':')."""
    return problem.split(":", 1)[0]
```

```python
"""Near-duplicate detection on user messages (MinHash over character 3-grams).

Punctuation and whitespace are ignored, so "明天台中會下雨嗎？" and
"明天台中會下雨嗎" are the same message. LSH proposes candidates; each is
confirmed with the MinHash Jaccard estimate before it counts.
"""

from __future__ import annotations

import unicodedata

from datasketch import MinHash, MinHashLSH

NUM_PERM = 128


def _normalise(text: str) -> str:
    return "".join(ch for ch in text if not unicodedata.category(ch).startswith(("P", "Z", "C")))


def _shingles(text: str, n: int = 3) -> set[str]:
    t = _normalise(text)
    if len(t) <= n:
        return {t}
    return {t[i : i + n] for i in range(len(t) - n + 1)}


def minhash(text: str) -> MinHash:
    m = MinHash(num_perm=NUM_PERM, seed=1)
    for sh in _shingles(text):
        m.update(sh.encode("utf-8"))
    return m


def dedupe(texts: list[str], threshold: float) -> list[int]:
    """Indices to keep: the first of every near-duplicate cluster survives."""
    lsh = MinHashLSH(threshold=threshold, num_perm=NUM_PERM)
    hashes: dict[str, MinHash] = {}
    keep: list[int] = []
    for i, text in enumerate(texts):
        m = minhash(text)
        if any(m.jaccard(hashes[k]) >= threshold for k in lsh.query(m)):
            continue
        key = str(i)
        lsh.insert(key, m)
        hashes[key] = m
        keep.append(i)
    return keep


def matches_any(texts: list[str], reference: list[str], threshold: float) -> list[int]:
    """Indices of `texts` that are near-duplicates of any `reference` text."""
    if not texts or not reference:
        return []
    lsh = MinHashLSH(threshold=threshold, num_perm=NUM_PERM)
    hashes: dict[str, MinHash] = {}
    for i, text in enumerate(reference):
        key = f"r{i}"
        hashes[key] = minhash(text)
        lsh.insert(key, hashes[key])
    hits = []
    for i, text in enumerate(texts):
        m = minhash(text)
        if any(m.jaccard(hashes[k]) >= threshold for k in lsh.query(m)):
            hits.append(i)
    return hits
```

`src/twagent/data/splits.py`：

```python
"""Group-wise train/valid/test split with a leakage guard (spec §5.3).

Every record of one scenario group -- (category, tools, style) -- goes to the
same split, and groups are assigned per category so each split sees every
category. The guard lives in `SplitResult.__post_init__`, as in
astro-vlm-lora: a leaky split object cannot be constructed.
"""

from __future__ import annotations

import random
from collections import defaultdict
from dataclasses import dataclass
from itertools import combinations
from typing import Any

from .dedup import matches_any
from .scenarios import Scenario


class DataLeakageError(ValueError):
    pass


def group_of(record: dict[str, Any]) -> str:
    return Scenario.from_dict(record["scenario"]).group


def assert_no_leakage(parts: dict[str, list[dict[str, Any]]], threshold: float) -> None:
    groups = {name: {group_of(r) for r in recs} for name, recs in parts.items()}
    for a, b in combinations(parts, 2):
        shared = groups[a] & groups[b]
        if shared:
            raise DataLeakageError(f"groups in both {a} and {b}: {sorted(shared)[:3]}")
        hits = matches_any(
            [r["message"] for r in parts[b]], [r["message"] for r in parts[a]], threshold
        )
        if hits:
            raise DataLeakageError(f"{len(hits)} {b} message(s) near-duplicate a {a} message")


@dataclass(frozen=True)
class SplitResult:
    train: list[dict[str, Any]]
    valid: list[dict[str, Any]]
    test: list[dict[str, Any]]
    threshold: float = 0.8

    def __post_init__(self) -> None:
        assert_no_leakage(
            {"train": self.train, "valid": self.valid, "test": self.test}, self.threshold
        )


def split_by_group(
    records: list[dict[str, Any]],
    *,
    seed: int,
    valid_frac: float = 0.1,
    test_frac: float = 0.1,
    threshold: float = 0.8,
) -> SplitResult:
    by_category: dict[str, dict[str, list[dict[str, Any]]]] = defaultdict(lambda: defaultdict(list))
    for r in records:
        by_category[r["scenario"]["category"]][group_of(r)].append(r)
    rng = random.Random(seed)
    train: list[dict[str, Any]] = []
    valid: list[dict[str, Any]] = []
    test: list[dict[str, Any]] = []
    for category in sorted(by_category):
        groups = by_category[category]
        names = sorted(groups)
        rng.shuffle(names)
        n = sum(len(v) for v in groups.values())
        n_test = n_valid = 0
        for name in names:
            recs = groups[name]
            if n_test < test_frac * n:
                test.extend(recs)
                n_test += len(recs)
            elif n_valid < valid_frac * n:
                valid.extend(recs)
                n_valid += len(recs)
            else:
                train.extend(recs)
    return SplitResult(train=train, valid=valid, test=test, threshold=threshold)
```

```python
"""raw trajectories -> filtered, deduplicated, split, formatted dataset.

    train.jsonl / valid.jsonl   rendered training examples (segments + loss mask)
    test_cases.jsonl            synthetic test set in the cases.py format
    data_report.json            counts, drop reasons, config

The synthetic test set is written as cases rather than trajectories: plan 3's
evaluation runs the model under test through `run_agent` and compares
against these expectations, exactly as it does for the handwritten set.
"""

from __future__ import annotations

import json
from collections import Counter
from pathlib import Path
from typing import Any

from twagent.cases import EXPECTED_DECISION, case_from_dict, validate_case
from twagent.render import render_chat

from .dedup import dedupe, matches_any
from .scenarios import Scenario
from .splits import SplitResult, group_of, split_by_group
from .validate import reason_key, validate_record


def to_training_example(record: dict[str, Any]) -> dict[str, Any]:
    return {
        "id": record["scenario"]["id"],
        "group": group_of(record),
        "category": record["scenario"]["category"],
        "segments": [{"text": s.text, "train": s.train} for s in render_chat(record["messages"])],
    }


def to_case(record: dict[str, Any]) -> dict[str, Any]:
    s = Scenario.from_dict(record["scenario"])
    case = {
        "id": s.id,
        "category": s.category,
        "today": s.today.isoformat(),
        "messages": [{"role": "user", "content": record["message"]}],
        "expected": {
            "decision": EXPECTED_DECISION[s.category],
            "calls": record["result"]["tool_calls"]
            if EXPECTED_DECISION[s.category] == "call"
            else [],
        },
        "fail_tools": {s.tools[0]: s.failure} if s.failure else {},
        "note": f"synthetic; group={s.group}",
    }
    problems = validate_case(case_from_dict(case))
    if problems:
        raise ValueError(f"{s.id}: synthetic case is invalid: {problems}")
    return case


def build_dataset(
    raw: list[dict[str, Any]],
    *,
    handwritten_messages: list[str],
    seed: int,
    threshold: float = 0.8,
) -> tuple[SplitResult, dict[str, Any]]:
    reasons: Counter[str] = Counter()  # a rejected record may count under several
    kept = []
    for record in raw:
        problems = validate_record(record)
        if problems:
            reasons.update({reason_key(p) for p in problems})
        else:
            kept.append(record)
    rejected = len(raw) - len(kept)

    contaminated = set(matches_any([r["message"] for r in kept], handwritten_messages, threshold))
    kept = [r for i, r in enumerate(kept) if i not in contaminated]

    unique = dedupe([r["message"] for r in kept], threshold)
    near_duplicates = len(kept) - len(unique)
    kept = [kept[i] for i in unique]

    split = split_by_group(kept, seed=seed, threshold=threshold)
    report = {
        "raw": len(raw),
        "kept": len(kept),
        "keep_rate": round(len(kept) / len(raw), 4) if raw else 0.0,
        # raw == kept + rejected + contaminated + near_duplicates
        "rejected": rejected,
        "contaminated": len(contaminated),
        "near_duplicates": near_duplicates,
        "reject_reasons": dict(sorted(reasons.items())),
        "splits": {
            name: {
                "records": len(part),
                "groups": len({group_of(r) for r in part}),
                "by_category": dict(
                    sorted(Counter(r["scenario"]["category"] for r in part).items())
                ),
            }
            for name, part in (("train", split.train), ("valid", split.valid), ("test", split.test))
        },
    }
    return split, report


def write_dataset(split: SplitResult, out_dir: Path) -> dict[str, Path]:
    out_dir.mkdir(parents=True, exist_ok=True)
    rows = {
        "train.jsonl": [to_training_example(r) for r in split.train],
        "valid.jsonl": [to_training_example(r) for r in split.valid],
        "test_cases.jsonl": [to_case(r) for r in split.test],
    }
    paths = {}
    for name, items in rows.items():
        path = out_dir / name
        with open(path, "w", encoding="utf-8", newline="\n") as f:
            for item in items:
                f.write(json.dumps(item, ensure_ascii=False) + "\n")
        paths[name] = path
    return paths
```

`src/twagent/data/commands.py`：

```python
"""`twagent synth` and `twagent build-dataset`.

    twagent synth --pilot 50                  # cost/keep-rate pilot -> data/synthetic/pilot.jsonl
    twagent synth --total 4000                # full run -> data/synthetic/raw.jsonl (resumable)
    twagent build-dataset                     # raw.jsonl -> train/valid/test_cases + report

The API key comes from the environment or a git-ignored `.env` file; this
code never prints or stores it. Spend is tracked in `<out>.cost.json` across
resumed runs, and a run stops before its batch would start past the budget.
"""

from __future__ import annotations

import argparse
import json
import math
import os
from pathlib import Path
from typing import Any

from twagent.cases import load_cases
from twagent.claude import AnthropicClient
from twagent.config import load_config

from .build import build_dataset, write_dataset
from .scenarios import build_scenarios
from .synth import load_records, synthesize
from .validate import validate_record

TRAIN_SHARE = 0.8  # split_by_group puts ~10% in valid and ~10% in test


def load_dotenv(path: Path) -> None:
    """Minimal KEY=VALUE loader; never overrides a variable already set."""
    if not path.exists():
        return
    for line in path.read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        key, value = line.split("=", 1)
        os.environ.setdefault(key.strip(), value.strip().strip('"').strip("'"))


def _read_ledger(path: Path) -> dict[str, Any]:
    if path.exists():
        return json.loads(path.read_text(encoding="utf-8"))
    return {"spent_usd": 0.0, "runs": []}


def cmd_synth(args: argparse.Namespace) -> int:
    cfg = load_config(args.config).synth
    load_dotenv(Path(".env"))
    model = args.model or cfg.model
    if model not in cfg.prices:
        print(f"ERROR no price for {model!r} in synth.prices; add it before spending money")
        return 2
    price = cfg.prices[model]
    if args.pilot:
        scenarios = build_scenarios(args.pilot, seed=cfg.seed + 1)
        out = Path(args.out or "data/synthetic/pilot.jsonl")
    else:
        scenarios = build_scenarios(args.total, seed=cfg.seed)
        out = Path(args.out or "data/synthetic/raw.jsonl")
    budget = args.budget if args.budget is not None else cfg.budget_usd

    teacher = AnthropicClient(
        model, max_tokens=cfg.max_tokens, disable_thinking=cfg.disable_thinking
    )
    query = AnthropicClient(
        model, max_tokens=cfg.max_tokens * 4, disable_thinking=cfg.disable_thinking
    )
    ledger_path = out.with_name(out.name + ".cost.json")
    ledger = _read_ledger(ledger_path)
    previous = ledger["spent_usd"]

    def spent() -> float:
        return previous + teacher.usage.cost_usd(price) + query.usage.cost_usd(price)

    report = synthesize(
        scenarios,
        query_llm=query,
        teacher=teacher,
        out_path=out,
        batch_size=cfg.query_batch_size,
        spent_usd=spent,
        budget_usd=budget,
    )
    ledger["spent_usd"] = round(spent(), 6)
    ledger["runs"].append(
        {
            "model": model,
            "written": report.written,
            "cost_usd": round(spent() - previous, 6),
            "teacher_usage": vars(teacher.usage),
            "query_usage": vars(query.usage),
        }
    )
    ledger_path.write_text(json.dumps(ledger, ensure_ascii=False, indent=2), encoding="utf-8")

    records = load_records(out)
    kept = sum(1 for r in records if not validate_record(r))
    keep_rate = kept / len(records) if records else 0.0
    per_record = ledger["spent_usd"] / len(records) if records else 0.0
    print(f"model {model} | wrote {report.written} | already had {report.already_done} | "
          f"missing queries {report.missing_queries} | stopped: {report.stopped}")  # fmt: skip
    print(f"records {len(records)} | keep rate {keep_rate:.1%} | spent ${ledger['spent_usd']:.2f} "
          f"| ${per_record:.4f} per record")  # fmt: skip
    if keep_rate > 0:
        needed = math.ceil(cfg.train_target / TRAIN_SHARE / keep_rate)
        projected = needed * per_record
        verdict = "within budget" if projected <= cfg.budget_usd else "EXCEEDS budget"
        print(f"projection for train_target={cfg.train_target}: {needed} raw records, "
              f"~${projected:.2f} with {model} ({verdict} ${cfg.budget_usd:.2f})")  # fmt: skip
    return 0


def cmd_build_dataset(args: argparse.Namespace) -> int:
    cfg = load_config(args.config)
    raw = load_records(Path(args.raw))
    if not raw:
        print(f"ERROR no records in {args.raw}")
        return 1
    handwritten: list[str] = []
    if Path(args.handwritten).exists():
        cases, _ = load_cases(args.handwritten)
        handwritten = [c.messages[-1]["content"] for c in cases]
    split, report = build_dataset(raw, handwritten_messages=handwritten, seed=cfg.synth.seed)
    out_dir = Path(args.out_dir)
    write_dataset(split, out_dir)
    report["handwritten_reference"] = len(handwritten)
    report["config"] = cfg.to_dict()
    (out_dir / "data_report.json").write_text(
        json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    print(f"raw {report['raw']} -> kept {report['kept']} ({report['keep_rate']:.1%}); "
          f"rejected {report['rejected']}, contaminated {report['contaminated']}, "
          f"near-duplicates {report['near_duplicates']}")  # fmt: skip
    for name, part in report["splits"].items():
        print(f"  {name}: {part['records']} records, {part['groups']} groups")
    return 0


def add_parsers(sub: Any, default_config: Path) -> None:
    p = sub.add_parser("synth", help="generate teacher trajectories with Claude")
    mode = p.add_mutually_exclusive_group(required=True)
    mode.add_argument("--pilot", type=int, help="run N pilot scenarios (separate seed and file)")
    mode.add_argument("--total", type=int, help="scenario count for the full run")
    p.add_argument("--model", help="override synth.model")
    p.add_argument("--budget", type=float, help="override synth.budget_usd for this run")
    p.add_argument("--out", help="output JSONL")
    p.add_argument("--config", type=Path, default=default_config)
    p.set_defaults(func=cmd_synth)

    p = sub.add_parser("build-dataset", help="validate, dedupe, split and format raw records")
    p.add_argument("--raw", default="data/synthetic/raw.jsonl")
    p.add_argument("--out-dir", default="data/synthetic")
    p.add_argument("--handwritten", default="data/handwritten_test.jsonl")
    p.add_argument("--config", type=Path, default=default_config)
    p.set_defaults(func=cmd_build_dataset)
```

```python
def _chat(args: argparse.Namespace) -> int:

def main(argv: list[str] | None = None) -> int:

    # cannot encode most of what this CLI prints (Traditional Chinese, emoji).

    # output and non-Windows platforms may not have `reconfigure`).

        sys.stderr.reconfigure(encoding="utf-8")

    sub = parser.add_subparsers(dest="command", required=True)

    p = sub.add_parser("check-cases", help="validate a test-case JSONL file")
    p.add_argument("path", type=Path)

    p = sub.add_parser("chat", help="run one message through the agent (mock tools)")
```
